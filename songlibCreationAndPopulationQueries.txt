CREATE TABLE category (
	categoryid integer NOT NULL PRIMARY KEY,
	categoryname varchar(64) NOT NULL UNIQUE
);

CREATE TABLE artist (
	artistid integer NOT NULL PRIMARY KEY,
	artistname varchar(64) NOT NULL UNIQUE
);

CREATE TABLE album (
	albumid integer NOT NULL PRIMARY KEY,
	albumtitle varchar(64),
	releaseyear integer
);

CREATE TABLE song (
	songid integer NOT NULL PRIMARY KEY,
	songtitle varchar(64)
);

CREATE TABLE isin (
	categoryid integer NOT NULL,
	songid integer NOT NULL,
	FOREIGN KEY (categoryid) REFERENCES category(categoryid),
	FOREIGN KEY (songid) REFERENCES song(songid),
	CONSTRAINT primarykey_isin PRIMARY KEY (categoryid, songid)
);

CREATE TABLE plays (
	artistid integer NOT NULL,
	songid integer NOT NULL,
	FOREIGN KEY (artistid) REFERENCES artist(artistid),
	FOREIGN KEY (songid) REFERENCES song(songid),
	CONSTRAINT primarykey_plays PRIMARY KEY (artistid, songid)
);

CREATE TABLE ison (
	albumid integer NOT NULL,
	songid integer NOT NULL,
	FOREIGN KEY (albumid) REFERENCES album(albumid),
	FOREIGN KEY (songid) REFERENCES song(songid),
	CONSTRAINT primarykey_ison PRIMARY KEY (albumid, songid)
);
